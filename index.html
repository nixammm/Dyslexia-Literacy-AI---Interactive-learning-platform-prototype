    // ─── Real-Time Web Speech API (TTS) Engine ───

    function speakText(text, rate = 0.9, pitch = 1.0) {

      if (!('speechSynthesis' in window)) {

        alert("Your browser does not support Web Speech API text-to-speech.");

        return;

      }

      

      window.speechSynthesis.cancel();

      

      const utterance = new SpeechSynthesisUtterance(text);

      utterance.rate = rate; // Optimized slower pacing for dyslexic auditory processing

      utterance.pitch = pitch;

      utterance.lang = 'en-US';

      

      const voices = window.speechSynthesis.getVoices();

      if (voices.length > 0) {

        const friendlyVoice = voices.find(v => v.lang.includes('en') && (v.name.includes('Google') || v.name.includes('Samantha') || v.name.includes('Natural') || v.name.includes('Victoria'))) || voices.find(v => v.lang.includes('en'));

        if (friendlyVoice) utterance.voice = friendlyVoice;

      }


      showAudioWave(text);


      utterance.onend = () => hideAudioWave();

      utterance.onerror = () => hideAudioWave();


      window.speechSynthesis.speak(utterance);

    }

    function initCanvas() {

      const canvas = document.getElementById('traceCanvas');

      canvas.width = canvas.offsetWidth;

      canvas.height = canvas.offsetHeight;

      ctx = canvas.getContext('2d');

      ctx.lineWidth = 10;

      ctx.lineCap = 'round';

      ctx.lineJoin = 'round';

      ctx.strokeStyle = '#4f7cff';


      canvas.onmousedown = (e) => { e.preventDefault(); ctx.beginPath(); const r = canvas.getBoundingClientRect(); ctx.moveTo(e.clientX-r.left, e.clientY-r.top); isDrawing = true; };

      canvas.onmousemove = (e) => { if (!isDrawing) return; const r = canvas.getBoundingClientRect(); ctx.lineTo(e.clientX-r.left, e.clientY-r.top); ctx.stroke(); };

      canvas.onmouseup = () => { isDrawing = false; ctx.closePath(); };

      canvas.onmouseout = () => { isDrawing = false; };


      canvas.ontouchstart = (e) => { e.preventDefault(); const t = e.touches[0]; const r = canvas.getBoundingClientRect(); ctx.beginPath(); ctx.moveTo(t.clientX-r.left, t.clientY-r.top); isDrawing = true; };

    function setTheme(btn, cls) {

      document.querySelectorAll('.theme-choice').forEach(b => b.classList.remove('active'));

    function applyFontFamily(fontName) {

      let fontStr = "'Inter', -apple-system, BlinkMacSystemFont, sans-serif";

      if (fontName === 'OpenDyslexic') {

        fontStr = "'OpenDyslexic', 'Comic Sans MS', 'Comic Neue', cursive, sans-serif";

      } else if (fontName === 'Lexend') {

        fontStr = "'Lexend', 'Inter', sans-serif";

      } else if (fontName === 'Comic Sans MS') {

        fontStr = "'Comic Sans MS', 'Comic Neue', cursive, sans-serif";

      } else if (fontName === 'Arial') {

        fontStr = "'Arial', 'Helvetica Neue', sans-serif";

      }

      

      document.documentElement.style.setProperty('--font', fontStr);

      document.body.style.fontFamily = fontStr;

      

      document.querySelectorAll('body, button, input, select, textarea, .word-text, .word-card, .btn, .nav-item, h1, h2, h3, h4, p, span, div').forEach(el => {

        if (!el.classList.contains('brand-icon') && !el.classList.contains('nav-icon') && !el.classList.contains('step-number')) {

          el.style.fontFamily = fontStr;

        }

      });


      showSettingToast(`✨ Font applied: ${fontName}`);

    }


    function applyLineSpacing(spacing) {

      document.body.style.lineHeight = spacing;

    }


    function setFontSize(btn, size) {

      document.querySelectorAll('.fs-btn').forEach(b => b.classList.remove('active'));

      if (btn) btn.classList.add('active');
